  // SkillSystemIntegration.Initialize() — 一键初始化的核心编排
  public void Initialize() {
    RegisterAllSkills();       // 注册28个内置技能
    RegisterStrategyHandlers(); // 关联6个策略处理器
    ValidateSkills();           // 验证引用完整性
    ConnectExecutorEvents();    // 连接执行引擎事件
  }

  // SkillExecutor.ExecuteAsync() — 递归执行核心逻辑
  public async Task<SkillExecutionResult> ExecuteAsync(skillId, params, ct) {
    var skill = Registry.GetSkill(skillId);
    return skill.Type switch {
      Basic => await ExecuteBasicAsync(skill, params, ct),
      Composite => await ExecuteCompositeAsync(skill, params, ct),
      Strategy => await ExecuteStrategyDecisionAsync(skill, params, ct),
    };
  }

  // 组合技能步骤循环+降级处理
  foreach (var step in skill.Steps) {
    var subResult = await ExecuteAsync(step.SkillId, mergedParams, ct);
    if (subResult.Status == Failed && step.FallbackSkillId != null) {
      subResult = await ExecuteAsync(step.FallbackSkillId, mergedParams, ct); // 降级
    } else if (subResult.Status == Failed && step.IsOptional) {
      continue; // 可选步骤跳过
    }
  }
}
